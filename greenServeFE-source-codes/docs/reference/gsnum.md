- sum
- product
- min
- max
- average
- mean
- pow
- sqrt
- abs
- floor
- ceil
- round
- sin
- cos
- tan
- mod
- clamp
- factorial

~~~gsve
import gsnum;

define values = [4, 8, 15, 16, 23, 42];

print(gsnum.sum(values));
print(gsnum.mean(values));
print(gsnum.min(4, 8, 15, 16, 23, 42));
print(gsnum.max(4, 8, 15, 16, 23, 42));
~~~
